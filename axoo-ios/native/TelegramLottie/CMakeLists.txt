cmake_minimum_required(VERSION 3.16)
project(AxooNativeLottie LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
add_subdirectory(../../vendor/TelegramRLottie telegram-engine)
add_library(axoo_native_lottie STATIC AxooLottie.cpp)
target_include_directories(axoo_native_lottie PUBLIC include)
target_link_libraries(axoo_native_lottie PUBLIC lottie_engine ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
add_executable(native_lottie_tests native_lottie_tests.cpp)
target_link_libraries(native_lottie_tests PRIVATE axoo_native_lottie)
enable_testing()
file(GLOB_RECURSE BUNDLED_TGS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../app/resources/axooji/*.tgs")
string(REPLACE ";" "\n" TGS_MANIFEST "${BUNDLED_TGS}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tgs-manifest.txt" "${TGS_MANIFEST}\n")
add_test(NAME native_lottie_tests COMMAND native_lottie_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../app/resources/axooji" "${CMAKE_CURRENT_BINARY_DIR}/tgs-manifest.txt")
